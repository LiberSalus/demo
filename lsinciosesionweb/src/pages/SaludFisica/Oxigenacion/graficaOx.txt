// src/components/Oxigenacion/GraficaOxigenacion.jsx
import React, { useState, useMemo, useEffect } from "react";
import styles from "./GraficaOxigenacion.module.css";
import {
  ResponsiveContainer,
  ScatterChart,
  Scatter,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  ReferenceLine,
} from "recharts";

// --- datos de ejemplo (igual que ya tenías) ---
const dataDia = [ /* ... */ ];
const dataSemana = [ /* ... */ ];
const dataMes = [ /* ... */ ];
const dataAnio = [ /* ... */ ];

const GraficaOxigenacion = () => {
  const [vista, setVista] = useState("dia");
  const [opcionSeleccionada, setOpcionSeleccionada] = useState("");

  // 🔹 Opciones del combo según la vista
  const opcionesFecha = useMemo(() => {
    switch (vista) {
      case "semana":
        return [
          { value: "semana-actual", label: "Semana actual" },
          { value: "semana-1", label: "Semana anterior" },
          { value: "semana-2", label: "Hace 2 semanas" },
        ];
      case "mes":
        return [
          { value: "2025-10", label: "Octubre 2025" },
          { value: "2025-09", label: "Septiembre 2025" },
          { value: "2025-08", label: "Agosto 2025" },
        ];
      case "anio":
        return [
          { value: "2025", label: "Año 2025" },
          { value: "2024", label: "Año 2024" },
          { value: "2023", label: "Año 2023" },
        ];
      case "dia":
      default:
        return [
          { value: "2025-10-21", label: "Mar 21, Oct 2025" },
          { value: "2025-10-20", label: "Lun 20, Oct 2025" },
          { value: "2025-10-19", label: "Dom 19, Oct 2025" },
        ];
    }
  }, [vista]);

  // 🔹 Cuando cambie la vista, aseguramos que la opción seleccionada
  // pertenezca a las opciones de esa vista
  useEffect(() => {
    if (!opcionesFecha.length) return;
    const existe = opcionesFecha.some((op) => op.value === opcionSeleccionada);
    if (!existe) {
      setOpcionSeleccionada(opcionesFecha[0].value);
    }
  }, [vista, opcionesFecha, opcionSeleccionada]);

  // 🔹 Config gráfica según la vista (igual que ya tenías, solo con labels mejores)
  const { data, xDomain, xTicks, xLabelFormatter } = useMemo(() => {
    switch (vista) {
      case "semana":
        return {
          data: dataSemana,
          xDomain: [1, 7],
          xTicks: [1, 2, 3, 4, 5, 6, 7],
          xLabelFormatter: (v) => {
            const dias = ["L", "M", "X", "J", "V", "S", "D"];
            return dias[v - 1] || v;
          },
        };
      case "mes":
        return {
          data: dataMes,
          xDomain: [1, 31],
          xTicks: [1, 5, 10, 15, 20, 25, 30],
          xLabelFormatter: (v) => v,
        };
      case "anio":
        return {
          data: dataAnio,
          xDomain: [1, 12],
          xTicks: [1, 3, 6, 9, 12],
          xLabelFormatter: (v) => {
            const meses = { 1: "Ene", 3: "Mar", 6: "Jun", 9: "Sep", 12: "Dic" };
            return meses[v] || v;
          },
        };
      case "dia":
      default:
        return {
          data: dataDia,
          xDomain: [0, 24],
          xTicks: [0, 6, 12, 18, 24],
          xLabelFormatter: (v) => v,
        };
    }
  }, [vista]);

  return (
    <div className={styles.card}>
      {/* Pills Día / Semana / Mes / Año */}
      <div className={styles.tabs}>
        <button
          type="button"
          className={`${styles.tab} ${vista === "dia" ? styles.tabActiva : ""}`}
          onClick={() => setVista("dia")}
        >
          Día
        </button>
        <button
          type="button"
          className={`${styles.tab} ${
            vista === "semana" ? styles.tabActiva : ""
          }`}
          onClick={() => setVista("semana")}
        >
          Semana
        </button>
        <button
          type="button"
          className={`${styles.tab} ${vista === "mes" ? styles.tabActiva : ""}`}
          onClick={() => setVista("mes")}
        >
          Mes
        </button>
        <button
          type="button"
          className={`${styles.tab} ${
            vista === "anio" ? styles.tabActiva : ""
          }`}
          onClick={() => setVista("anio")}
        >
          Año
        </button>
      </div>

      {/* Gráfica */}
      <div className={styles.chartArea}>
        <ResponsiveContainer width="100%" height="100%">
          <ScatterChart margin={{ top: 10, right: 10, bottom: 10, left: 0 }}>
            <CartesianGrid strokeDasharray="3 3" />

            <XAxis
              type="number"
              dataKey="x"
              domain={xDomain}
              ticks={xTicks}
              tickFormatter={xLabelFormatter}
              className={styles.XAxis}
            />

            <YAxis
              type="number"
              dataKey="spo2"
              domain={[70, 100]}
              ticks={[70, 75, 80, 85, 90, 92, 94, 96, 98, 100]}
              tickFormatter={(v) => `${v}%`}
              className={styles.YAxis}
            />

            <ReferenceLine y={92} stroke="#16a34a" strokeWidth={2} strokeDasharray="3 0" />
            <ReferenceLine y={85} stroke="#fbbf24" strokeWidth={2} strokeDasharray="4 4" />
            <ReferenceLine y={75} stroke="#f97373" strokeWidth={2} strokeDasharray="3 0" />

            <Tooltip
              formatter={(value) => `${value}%`}
              labelFormatter={(v) => `Punto: ${v}`}
            />

            <Scatter
              data={data}
              fill="#0284c7"
              stroke="#0284c7"
              strokeWidth={1.4}
            />
          </ScatterChart>
        </ResponsiveContainer>
      </div>

      {/* Combo de fecha + Ver registros */}
      <div className={styles.bottomBar}>
        <div className={styles.fechaSelectWrapper}>
          <select
            className={styles.fechaSelect}
            value={opcionSeleccionada}
            onChange={(e) => setOpcionSeleccionada(e.target.value)}
          >
            {opcionesFecha.map((op) => (
              <option key={op.value} value={op.value}>
                {op.label}
              </option>
            ))}
          </select>
        </div>

        <button type="button" className={styles.linkReg}>
          Ver registros
        </button>
      </div>
    </div>
  );
};

export default GraficaOxigenacion;
